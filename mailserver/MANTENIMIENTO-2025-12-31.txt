================================================================================
11. COMANDOS UTILES PARA FUTURO
================================================================================

FAIL2BAN
--------
# Ver estado general
sudo docker exec mailserver fail2ban-client status

# Ver jail especifica
sudo docker exec mailserver fail2ban-client status dovecot

# Desbanear IP
sudo docker exec mailserver fail2ban-client set dovecot unbanip <IP>

# Ver whitelist
sudo docker exec mailserver fail2ban-client get dovecot ignoreip

# Desbanear todas las IPs
sudo docker exec mailserver fail2ban-client unban --all

POSTFIX UTF-8
-------------
# Ver configuracion SMTP UTF-8
sudo docker exec mailserver postconf smtputf8_enable

# Ver configuracion MIME
sudo docker exec mailserver postconf | grep mime

# Recargar Postfix
sudo docker exec mailserver postfix reload

# Ver configuracion completa
sudo docker exec mailserver postconf -n

LOGS
----
# Ver logs en tiempo real
sudo docker logs -f mailserver

# Buscar errores
sudo docker logs mailserver --since 24h | grep -iE "error|failed|warning"

# Ver logs de usuarios especificos
sudo docker logs mailserver | grep -E "facturacion@|finanzas@"

CONTENEDOR
----------
# Reiniciar contenedor
sudo docker restart mailserver

# Reiniciar con compose
sudo docker compose down
sudo docker compose up -d

# Ver estado
sudo docker ps -a | grep mailserver

# Entrar al contenedor
sudo docker exec -it mailserver bash
